
LootJS.lootTables(event => {
    event
        .getLootTable("minecraft:chests/desert_pyramid")
        .firstPool()
        .addEntry(LootEntry.of("minecraft:wither_skeleton_skull").withWeight(20).setCount([1, 50]))
})

LootJS.lootTables(event => {
    let ids = event.getLootTableIds()
    console.log(ids)
    ids.forEach(element => {
        event.getLootTable(element).removeItem('relics:roller_skates')
    });

    event.getLootTable('minecraft:chests/desert_pyramid').removeItem('relics:roller_skates')
})


LootJS.modifiers(event => {
    console.log(event.getGlobalModifiers())
    event.removeGlobalModifiers(/relics:.*/)

})
